\begin{proc}{registrarUsuario}{\Inout S: AirMiles, \In id:IdUsuario}{}

\begin{proc}{acumularMillas}{
\Inout s : AirMiles, 
\In id : IdUsuario, 
\In d : \R, 
\In p : \Tipo{NombrePromo} }
{}
    \requiere{
        s = s_0 \land \\
        d > 0 \land \\
        usuarioRegistrado(id, s) \land \\
        p \in claves(s.promoSinTope) \lor p\in claves(s.promoConTope) \land \\
        ((p \in claves(s.promoConTope) \implicaLuego millasValidasParaPromo(id, d, p, s))}
    \medskip
        \asegura{
            s.promoSinTope = s_0.promoSinTope \land \\
            \\ 
            \IfThenElse{(p \in claves(s_0.promoConTope)) \land \\ 
            (sumarMillasAlTope(id, d, p, s_0) = s_0.promoConTope[p].tope))}{ \\
            \\
            s.promoConTope = delKey(s_0.promoConTope, p)}{\\
            \IfThenElse{(p \in claves(s_0.promoConTope))}{\\
            \\
            claves(s.promoConTope) = claves(s_0.promoConTope) \land \\
            s.promoConTope[p].millasAcumuladas = sumarMillasAlTope(id, d, p, s_0) \land \\
            s.promoConTope[p].tope = s_0.promoConTope[p].tope \land \\
            (\forall p' \in claves(s_0.promoConTope))(p'\neq p \implicaLuego s.promoConTope[p'] = s_0.promoConTope[p'])}{ \\
            \\
            (s.promoConTope = s_0.promoConTope)}} \\
            \\
            |s.usuario[id].vuelos| = |s_0.usuario[id].vuelos| + 1 \land \\
            (\forall i : \Z)(0 \leq i < |s_0.usuario[id].vuelos| \implicaLuego s.usuario[id].vuelos[i] = s_0.usuario[id].vuelos[i]) \land 
            s.usuario[id].vuelos[|s_0.usuario[id].vuelos|] = nuevoVuelo(id, d, p, s_0.promoConTope[p].factor) \oLuego s.usuario[id].vuelos[|s_0.usuario[id].vuelos|] = nuevoVuelo(id, d, p, s_0.promoSinTope[p].factor) \land
            s.usuario[id].canjes = s_0.usuario[id].canjes \land \\
            s.usuario[id].transferenciasRecibidas = s_0.usuario[id].transferenciasRecibidas \land \\
            s.usuario[id].transferenciasEnviadas = s_0.usuario[id].transferenciasEnviadas \land \\
            s.usuario[id].referidos = s_0.usuario[id].referidos \land \\
            (\forall id' \in claves(s.usuario))(id' \neq id) \implicaLuego s.usuario[id'] = s_0.usuario[id'])}

\end{proc}

\aux{factorCategoria}{millasTotales: \R}{\R}{
        \IfThenElse{millasTotales < 10000}{1.0}{ \\
            \IfThenElse{millasTotales < 50000}{1.5}{ \\

\aux{calcularMillasGanadas}{id : IdUsuario, d : \R, fp : \R, s : \Tipo{AirMiles}}{\R}{

}

\begin{proc}{canjearMillas}{
    \In id: idUsuario,
    \In millasCanjeadas: \R,
    \In premio: \str,
    \Inout s: AirMiles
}{}
    \requiere{
        s = s_0 \ \land \\
        usuarioRegistrado(id, s) \ \land_L \\
        millasDisponibles(id, s) \geq millasCanjeadas
    }
    
    \asegura{
        s.promoSinTope = s_0.promoSinTope \ \land \\
        s.promoConTope = s_0.promoConTope \ \land \\
        \claves{s.usuario} = \claves{s_0.usuario} \ \land \\
        millasTotales(id, s) = millasTotales(id, s_0) \ \land \\
        millasDisponibles(id, s) = millasDisponibles(id, s_0) - millasCanjeadas \ \land \\
        s.usuario[id].canjes \concat \literal{millas: millasCanjeadas, premio: premio} = s_0.usuario[id].canjes
    }

% 1. Auxiliar de Millas Por Vuelo (Suma de millas del historial de vuelos)



\medskip

\aux{millasPorVuelos}{id: IdUsuario, s: AirMiles}{\R}{
    \sum_{i=0}^{|S.usuario[id].vuelos| - 1} s.usuario[id].vuelos[i].millas
}

\medskip

\aux{millasPorCanjes}{id: IdUsuario, s: AirMiles}{\R}{
    \sum_{i=0}^{|S.usuario[id].Canjes| - 1} s.usuario[id].vuelos[i].millas
}

\medskip

\aux{millasPorTransferenciaEnviadas}{id: IdUsuario, s: AirMiles}{\R}{
    \sum_{i=0}^{|S.usuario[id].TransferenciasEnviadas| - 1} s.usuario[id].TransferenciasEnviadas[i].millas
}

\medskip

\aux{millasPorTransferenciaRecibidas}{id: IdUsuario, s: AirMiles}{\R}{
    \sum_{i=0}^{|S.usuario[id].TransferenciasRecibidas| - 1} s.usuario[id].TransferenciasRecibidas[i].millas
}

\medskip

\aux{millasPorReferidOrigens}{id: IdUsuario, s: AirMiles}{\R}{
    \sum_{i=0}^{|S.usuario[id].ReferidOrigens| - 1} s.usuario[id].ReferidOrigens[i].millas
}

\medskip

% 2. Auxiliar de Millas Totales (Millas por Vuelo + Transferencias Recibidas)
\aux{millasTotales}{id:IdUsuario, s: AirMiles}{\R}{
    millasPorVuelo(id, s) + millasPorTransferenciaRecibida(id, s) + millasPorReferidOrigens(id, s)
}

\medskip
\medskip

% 3. Millas Perdidas (Transferencias Enviadas - Canjes)
\aux{millasPerdidas}{id: IdUsuario, s: AirMiles}{\R}{
    millasPorTransferenciaEnviadas(id, s) + millasPorCanje(id, s)
}

\medskip

% 3. Auxiliar de Millas Disponibles
\aux{millasDisponibles}{id: IdUsuario, s: AirMiles}{\R}{
    millasTotales(id, s) - millasPerdidas(id, s)
}

\end{proc}

\begin{proc}{transferirMillas}{
\Inout s : AirMiles,
\In idOrigen : IdUsuario,
\In idDestino : IdUsuario,\\
\In millasTransferidas : \R
}
{}
    \requiere{
        s = s_0 \land \\
        UsuarioRegistrado(idOrigen,s) \yLuego \\
        UsuarioRegistrado(idDestino,s) \yLuego \\
        idOrigen \neq idDestino \land \\
        millasTransferidas > 0 \land \\
        millasDisponibles(idOrigen, s) \geq millasTransferidas}

    \asegura{
        claves(s.usuario) = claves(s_0.usuario) \yLuego\\
        claves(s.promoSinTope) = claves(s_0.promoSinTope) \yLuego \\
        claves(s.promoConTope) = claves(s_0.promoConTope)\\
        \\
        s.usuario[idOrigen].transferenciasEnviadas = \\
        s_0.usuario[idOrigen].transferenciasEnviadas ++ \\
        \langle \langle
        \text{millas: } millasTransferidas, 
        \text{usuarioOrigen: } idOrigen,
        \text{usuarioDestino: } idDestino
        \rangle \rangle \land \\
        s.usuario[idOrigen].vuelos = s_0.usuario[idOrigen].vuelos \land \\
        s.usuario[idOrigen].canjes = s_0.usuario[idOrigen].canjes \land \\
        s.usuario[idOrigen].transferenciasRecibidas = s_0.usuario[idOrigen].transferenciasRecibidas \land 
        s.usuario[idOrigen].referidos = s_0.usuario[idOrigen].referidos\\
        \\
        s.usuario[idDestino].transferenciasRecibidas = \\
        s_0.usuario[idDestino].transferenciasRecibidas ++ \\
        \langle \langle
        \text{millas: } millasTransferidas, 
        \text{usuarioOrigen: } idOrigen,
        \text{usuarioDestino: } idDestino
        \rangle \rangle \land \\
        s.usuario[idDestino].vuelos = s_0.usuario[idDestino].vuelos \land \\
        s.usuario[idDestino].canjes = s_0.usuario[idDestino].canjes \land \\
        s.usuario[idDestino].transferenciasEnviadas = s_.usuario[idDestino].transferenciasEnviadas \land
        s.usuario[idDestino].referidos = s_0.usuario[idDestino].referidos \land \\
        \\
        (\forall u : IdUsuario)(UsuarioRegistrado(u,s) \yLuego u \neq idOrigen \land 
        u \neq idDestino \implicaLuego s.usuario[u] = s_0.usuario[u])\\}

\end{proc}

\begin{proc}{rankingMillas}{\In n: \N, \In s: AirMiles}{\seq{idUsuario}}
    \requiere{
        n \le |\claves{s.usuario}|
    }
    \asegura{
        NoSeRepitenUsuarios(res) \ \land \\
        |res| = n \ \land \\
        (\forall u: idUsuario) \ (u \in res \implies_L usuarioRegistrado(u, s)) \ \land \\
        estaOrdenadaDecreciente(res, s) \ \land \\
        sonLosUsuariosConMasMillas(res, s)
    }


\medskip

\pred{estaOrdenadaDecreciente}{res: \seq{idUsuario}, s: AirMiles}{
    (\forall i: \Z) \ (0 \le i < |res| - 1 \implies_L \\
    \quad millasDisponibles(res[i], s) \ge millasDisponibles(res[i+1], s))
}

\medskip
\pred{noSeRepitenUsuarios}{res: \seq{idUsuario}}{
    (\forall u: \idUsuario) \ (u \in res \implies_L apariciones(u, res) = 1)
}

\medskip

\aux{apariciones}{u: \idUsuario, res: \seq{\idUsuario}}{\Z}{
    \sum_{i=0}^{|res|-1} \IfThenElse{res[i] = u}{1}{0}
}


\medskip
\pred{sonLosUsuariosConMasMillas}{res: \seq{idUsuario}, s: AirMiles}{
    (\forall u: idUsuario) \ ((usuarioRegistrado(u, s) \ \land \ u \notin res) \implies_L \\
    \quad \quad millasDisponibles(u, s) \le millasDisponibles(res[|res|-1], s))
}


\end{proc}


\begin{proc}{resumenTransacciones}{\In id: idUsuario, \In s: AirMiles}{\tupla{\R, \R, \R}}
    \requiereCorto{usuarioRegistrado(id, s)}
    \asegura{
        res_0 = millasTotalesPorVuelo(id, s) \ \land \\
        res_1 = millasTotalesPorCanje(id, s) \ \land \\
        res_2 = millasPorTransferenciaRecibida(id, s) + millasTotalesPorTransferenciaEnviada(id, s)
    }
\end{proc}

\begin{proc}{iniciarPromocion}{
\In p: \Tipo{NombrePromo}, 
\In factor: \Z,
\Inout s: \Tipo{AirMiles}}
{}
    \requiere{
        s = s_0 \land \\
        p \notin \claves{s.\text{promoSinTope}} \land p \notin claves(s.promoConTope) \land \\
        factor \ge 1.0 \\}

    \medskip

    \asegura{
        s.usuario = s_0.usuario \land \\
        s.promoConTope = s_0.promoConTope \land \\
        claves(s.promoSinTope) = claves(s_0.promoSinTope) \cup \{p\} \yLuego \\
        s.promoSinTope[p] =
        \langle 
            \text{millasAcumuladas: } 0.0, 
            \text{factor: } factor 
            \rangle  \land \\
        (\forall p' : NombrePromo)(p' \in claves(s_0.promoSinTope) \implicaLuego s.promoSinTope[p'] = s_0.promoSinTope[p'])}
        
\end{proc}

\begin{proc}{iniciarPromoConTope}{
\In p: NombrePromo, 
\In factor: \Z, 
\In tope: \R, 
\Inout s: AirMiles}{}
    \requiere{
            s = s_0 \land \\
            p \notin claves(s.promoSinTope) \land p \notin claves(s.promoConTope) \land \\
            factor \ge 1.0 \land \\
            tope > 0}
    \asegura{
            s.usuario = s_0.usuario \land \\
            s.promoSinTope = s_0.promoSinTope \land \\
            claves(s.promoConTope) = claves(s_0.promoConTope) \cup \{p\} \yLuego \\
            s.promoConTope[p] = 
            \langle 
            \text{millasAcumuladas: } 0.0, 
            \text{tope: } tope, 
            \text{factor: } factor 
            \rangle \land \\
            (\forall p' : NombrePromo)(p' \in claves(s_0.promoConTope) \implicaLuego s.promoConTope[p'] = s_0.promoConTope[p'])}
\end{proc}

\begin{proc}{finalizarPromocion}{\Inout s: AirMiles, \In p: NombrePromo}{}
    \requiere{
        s = s_0 \ \land_L \\
        p \neq \text{"NoPromo"} \ \land_L \\
        promoActiva(p, s_0)
    }
    \asegura{
        \begin{aligned}[t]
            &(p \in \claves{s_0.promoSinTope} \implies_L \\
            &\quad s.promoSinTope = \delKey{s_0.promoSinTope}{p} \ \land \ s.promoConTope = s_0.promoConTope) \ \land \\
            &(p \in \claves{s_0.promoConTope} \implies_L \\
            &\quad s.promoConTope = \delKey{s_0.promoConTope}{p} \ \land \ s.promoSinTope = s_0.promoSinTope) \ \land \\
            &s.usuario = s_0.usuario
        \end{aligned}
    }
    \end{proc}
\medskip
    
\pred{promoActiva}{nombre: NombrePromo, s: AirMiles}{
    nombre \in \claves{s.promoSinTope} \lor nombre \in \claves{s.promoConTope}
    }

    }
\end{proc}

\begin{proc}{paresTransferidoresExclusivos}{\In s: AirMiles}{\seq{\tupla{\idUsuario, \idUsuario}}}
    \asegura{
